# JARVIS backend dependencies — install into a venv at ./venv:
#   python -m venv venv
#   venv\Scripts\pip install -r requirements.txt   (Windows)
#   venv/bin/pip install -r requirements.txt        (macOS/Linux)
#
# This is the lean core (~400 MB smaller than before). Market scanner, video
# understanding, Chroma vectors and the Mem0 mirror live in requirements-extras.txt;
# each one degrades gracefully (with an install hint) when its package is absent.

# Web server
fastapi
uvicorn[standard]
websockets

# Brain / vision / STT — all via Groq's free OpenAI-compatible API
openai

# Text-to-speech (Microsoft Edge neural voices, free, cloud)
edge-tts

# Voice input capture + VAD
sounddevice
numpy

# Robust speech detection (spectral VAD) — separates speech from mic/background noise far
# better than an energy threshold. Prebuilt wheels, no C compiler needed. If absent, voice
# capture degrades to a simple energy gate rather than breaking.
webrtcvad-wheels

# Local speech-to-text (CTranslate2 Whisper, tiny.en, CPU int8) — the default STT backend.
# Runs offline with zero network round-trip, so wake-word detection is instant and doesn't
# depend on internet/Groq latency. Falls back to Groq cloud Whisper if this is missing.
faster-whisper

# System tools
psutil

# Screen capture (for cloud vision)
mss
Pillow

# Document uploads — PDF text extraction (docx/text use stdlib only)
pypdf

# Cross-model memory hub — MCP server so Claude/ChatGPT/Gemini share JARVIS's store
mcp<2   # 2.x renamed FastMCP; jarvis.agent_mcp + memory hub use the v1 API

# Local compute layer (governor/device/consolidation/models_advisor) talks to Ollama
ollama

# Mouse / keyboard control for the `desktop` tool (mouse_click, type_text, key_press …)
pyautogui

# Native OS toast notifications for `desktop.notify` (Windows Action Center / macOS /
# Linux libnotify — one API, cross-platform).
plyer

# Web search — optional faster DDG client (falls back to HTML scrape)
duckduckgo-search

# Optional fallback brain (only if you set ANTHROPIC_API_KEY)
anthropic

# Hermes-style skills (skills.py) — YAML frontmatter for SKILL.md. Explicit so boot
# does not depend on chromadb/ctranslate2 transitively providing PyYAML.
PyYAML
